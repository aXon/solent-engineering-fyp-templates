\chapter{Health, Safety and Ethics}
\label{app:hse}

\begin{guidance}[Required evidence (Handbook, Ethics and Health \& Safety)]
The Final Report submission must include:
\begin{itemize}
  \item the DSE self-assessment using the HSE checklist \parencite{hse2013dse};
  \item any GRA01 risk assessments for practical work, approved by your
    supervisor and the technical staff;
  \item the ethics release from the Solent Ethics App, and the Full Ethical
    Review approval if one was needed.
\end{itemize}
Save the PDFs in \texttt{evidence/} with the file names below; they are
included automatically.
\end{guidance}
