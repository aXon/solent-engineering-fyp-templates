\chapter{Implementation and Testing}

\begin{guidance}[What this chapter does]
Describes how the selected solution was built, modelled or carried out, and
how it was tested. Give enough detail (apparatus, settings, procedures) for
another engineer to repeat the work. Large drawings, full code and data
belong in the digital evidence (\cref{app:digital}), not in the text.
Typical length: 1,500--2,000 words.
\end{guidance}

import time
from sensor import read_temperature   # own driver, see digital evidence

# Lines 3-6 generated with GitHub Copilot, reviewed by the author
def log_sample(file, interval_s):
    t = read_temperature()
    file.write(f"{time.time():.0f},{t:.3f}\n")
    time.sleep(interval_s)
\end{lstlisting}
